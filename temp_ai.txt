// Fallback for crypto.randomUUID() on non-HTTPS mobile connections
const generateId = () => {
  if (typeof crypto !== 'undefined' && crypto.randomUUID) {
    return crypto.randomUUID();
  }
  return 'id_' + Date.now().toString(36) + '_' + Math.random().toString(36).substring(2);
}

// Build full RAG context including drive files
async function buildFullContext(): Promise<string> {
  const base = await buildRAGContext()

  // Add drive files
  const driveFiles = await db.driveFiles.toArray()
  const subjects = await db.subjects.toArray()
  const subjectMap = Object.fromEntries(subjects.map((s) => [s.id, s.name]))

  if (driveFiles.length > 0) {
    let filesCtx = '\n### ملفات Google Drive المتاحة\n'
    const grouped: Record<string, typeof driveFiles> = {}
    for (const f of driveFiles) {
      const key = f.subjectId ? subjectMap[f.subjectId] ?? 'غير مصنف' : 'غير مصنف'
      if (!grouped[key]) grouped[key] = []
      grouped[key].push(f)
    }
    for (const [subject, files] of Object.entries(grouped)) {
      filesCtx += `\n**${subject}**:\n`
      files.forEach((f) => {
        filesCtx += `- ${f.name} (${f.mimeType.split('/').pop()})\n`
      })
    }
    return base + filesCtx
  }

  return base
}

// Extract base64 attachments from tasks for multimodal
async function getTaskAttachments() {
  const tasks = await db.tasks.filter((t) => !!(t.attachments && t.attachments.length > 0)).toArray()
  const parts: { inlineData: { mimeType: string; data: string }; taskTitle: string; fileName: string }[] = []

  for (const task of tasks) {
    for (const att of task.attachments ?? []) {
      if (att.data) {
        // data is base64 data URL: "data:mime/type;base64,xxxx"
        const match = att.data.match(/^data:([^;]+);base64,(.+)$/)
        if (match) {
          parts.push({
            inlineData: { mimeType: match[1], data: match[2] },
            taskTitle: task.title,
            fileName: att.name,
          })
        }
      }
    }
  }
  return parts
}

export default function AIScreen() {
  const { geminiApiKey } = useSettingsStore()
  const { showToast, setActiveTab } = useUIStore()
  
  // Session state

  const [showSessionsMenu, setShowSessionsMenu] = useState(false)

  const [editTitle, setEditTitle] = useState('')

  const [loading, setLoading] = useState(false)

    () => db.messages.where('sessionId').equals(currentSessionId).sortBy('timestamp'),
    [currentSessionId]
  )

    localStorage.setItem('lastChatSessionId', currentSessionId)

    const existing = await db.chatSessions.get(currentSessionId)
    if (!existing) {
      await db.chatSessions.add({
        id: currentSessionId,
        title,
        createdAt: Date.now(),
        updatedAt: Date.now()

      await db.chatSessions.update(currentSessionId, { updatedAt: Date.now() })

    setCurrentSessionId(generateId())
    setShowSessionsMenu(false)

    e.stopPropagation()

    await db.chatSessions.delete(id)
    if (currentSessionId === id) {
      const remaining = sessions.filter(s => s.id !== id)

    }

    e.stopPropagation()
    setEditingSessionId(session.id)
    setEditTitle(session.title)
  }

  const saveRename = async (id: string, e?: React.FormEvent) => {
    e?.preventDefault()
    if (editTitle.trim()) {
      await db.chatSessions.update(id, { title: editTitle.trim(), updatedAt: Date.now() })
    }
    setEditingSessionId(null)
  }

    if (window.confirm('هل أنت متأكد من حذف هذه المحادثة؟')) {
      await db.messages.where('sessionId').equals(currentSessionId).delete()
      await db.chatSessions.delete(currentSessionId)
      createNewSession()
    }
  }

  const hasMessages = messages && messages.length > 0

  const sendMessage = async (text: string = input) => {
    if (!text.trim() || loading) return
    if (!geminiApiKey) {
      showToast('أضف Gemini API Key في الإعدادات أولاً', 'error')

      return

    // Ensure the session exists and create a title if it's the first message

      const title = userMsgText.length > 25 ? userMsgText.substring(0, 25) + '...' : userMsgText
      await ensureSessionExists(title)
    } else {
      await ensureSessionExists()
    }

      role: 'user',
      content: userMsgText,
      timestamp: Date.now(),
      sessionId: currentSessionId,
    }
    await db.messages.add(userMsg)
    setInput('')
    setLoading(true)

    try {

      const context = await buildFullContext()

      // Build chat history

        role: m.role === 'assistant' ? 'model' : ('user' as 'user' | 'model'),
        parts: [{ text: m.content }],
      }))

      const genAI = new GoogleGenerativeAI(geminiApiKey)

        systemInstruction: `أنت مساعد ذكي لطالب هندسة صناعية. مهمتك مساعدته في التنظيم والمذاكرة وفهم المواد.

${context}

قواعد:
- أجب باللغة العربية دائماً ما لم يطلب غير ذلك
- استخدم البيانات أعلاه للإجابة على الأسئلة التنظيمية بدقة
- عند شرح مواضيع دراسية استخدم أمثلة عملية من الهندسة الصناعية
- كن موجزاً ومفيداً. استخدم emoji باعتدال.`,

        role: 'assistant',
        content: responseText,
        timestamp: Date.now(),
        sessionId: currentSessionId,
      })

        sessionId: currentSessionId,
      })

      setLoading(false)

  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {

      <div className="flex items-center justify-between px-4 pt-5 pb-3 border-b border-surface-border flex-shrink-0 z-20">

                        <input
